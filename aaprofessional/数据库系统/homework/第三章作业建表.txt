CREATE DATABASE supply;
USE supply;

DROP TABLE IF EXISTS S;

CREATE TABLE S
(
    SNO     CHAR(2)         COMMENT '供应商代码' PRIMARY KEY ,
    SNAME   VARCHAR(10)     COMMENT '供应商姓名' UNIQUE ,
    STATUS  INT             COMMENT '供应商状态',
    CITY    VARCHAR(10)     COMMENT '供应商所在城市'
) COMMENT '供应商';

DROP TABLE IF EXISTS P;

CREATE TABLE P
(
    PNO     CHAR(2)         COMMENT '零件代码 ' PRIMARY KEY ,
    PNAME   VARCHAR(10)     COMMENT '零件名' ,
    COLOR   CHAR            COMMENT '颜色',
    WEIGHT  INT             COMMENT '重量'
) COMMENT '零件';

CREATE TABLE J
(
    JNO     CHAR(2)         COMMENT '工程项目代码 ' PRIMARY KEY ,
    JNAME   VARCHAR(10)     COMMENT '工程项目名' UNIQUE ,
    CITY    VARCHAR(10)     COMMENT '工程项目所在城市'
) COMMENT '工程项目';

CREATE TABLE SPJ
(
    SNO     CHAR(2)         COMMENT '供应商代码',
    PNO     CHAR(2)         COMMENT '零件代码' ,
    JNO     CHAR(2)         COMMENT '工程项目代码',
    QTY     INT             COMMENT '供应数量',
    FOREIGN KEY (SNO) REFERENCES S(SNO) ON UPDATE CASCADE ,
    FOREIGN KEY (PNO) REFERENCES P(PNO) ON UPDATE CASCADE ,
    FOREIGN KEY (JNO) REFERENCES J(JNO) ON UPDATE CASCADE
) COMMENT '供应情况';

SELECT SNO
FROM SPJ
WHERE JNO = 'J1';

SELECT SNO
FROM SPJ
WHERE JNO = 'J1' AND PNO = 'P1';

SELECT SNO
FROM SPJ
JOIN P ON SPJ.PNO = P.PNO
WHERE SPJ.JNO = 'J1' AND P.COLOR = '红';

SELECT DISTINCT SPJ.JNO
FROM SPJ
JOIN P
    ON P.PNO = SPJ.PNO
JOIN S
    ON SPJ.SNO = S.SNO
WHERE NOT (P.COLOR = '红' AND S.CITY = '天津');

SELECT DISTINCT JNO
FROM SPJ
WHERE JNO IN(
    SELECT JNO
    FROM SPJ
    WHERE PNO IN(
        SELECT PNO
        FROM SPJ
        WHERE SNO = 'S1'
        )
    )
