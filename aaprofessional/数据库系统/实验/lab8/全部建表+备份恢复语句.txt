DROP TABLE IF EXISTS student_instructor;
CREATE TABLE student_instructor
(
    stu_id      int      NOT NULL       COMMENT '学号'    PRIMARY KEY ,
    cs_id       int      NOT NULL       COMMENT '班级号',
    instructor  varchar(20)   NOT NULL  COMMENT '辅导员姓名',
    FOREIGN KEY student_instructor(stu_id) REFERENCES student(stu_id),
    FOREIGN KEY student_instructor(cs_id) REFERENCES class(cs_id)
)COMMENT '学生-辅导员 2NF';

# 触发器
ALTER TABLE student ADD COLUMN AvgGrade DECIMAL(4,1)
    DEFAULT 0.0 COMMENT '平均成绩';

UPDATE student s
SET s.AvgGrade = (
    SELECT AVG(grade)
    FROM take_course tc
    WHERE tc.stu_id = s.stu_id
)
WHERE EXISTS (SELECT 1 FROM take_course tc WHERE tc.stu_id = s.stu_id);

# 录入成绩时的触发器
DELIMITER //
CREATE TRIGGER trg_after_insert_grade
AFTER INSERT ON take_course
FOR EACH ROW
BEGIN
    UPDATE student
    SET AvgGrade = (SELECT AVG(grade) FROM take_course
                                      WHERE stu_id = NEW.stu_id)
    WHERE stu_id = NEW.stu_id;
END //

# 修改成绩时的触发器
CREATE TRIGGER trg_after_update_grade
AFTER UPDATE ON take_course
FOR EACH ROW
BEGIN
    -- 只有当成绩字段发生变化时才更新，节省开销
    IF OLD.grade <> NEW.grade THEN
        UPDATE student
        SET AvgGrade = (SELECT AVG(grade) FROM take_course
                                          WHERE stu_id = NEW.stu_id)
        WHERE stu_id = NEW.stu_id;
    END IF;
END //
DELIMITER ;
