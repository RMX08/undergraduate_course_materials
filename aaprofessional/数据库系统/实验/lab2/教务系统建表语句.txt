CREATE DATABASE education_system;
USE education_system;

DROP TABLE IF EXISTS college;

CREATE TABLE college
(
    col_id      int          NOT NULL   COMMENT '学院编号' PRIMARY KEY ,
    col_name    varchar(20)  NOT NULL   COMMENT '学院名称' UNIQUE ,
    campus      varchar(20)  NOT NULL   COMMENT '所在校区'
)comment '学院';

CREATE TABLE teaching_and_research_office
(
    tr_id     int           NOT NULL       COMMENT '教研室编号' PRIMARY KEY ,
    tr_name   varchar(20)   NOT NULL       COMMENT '教研室名' UNIQUE ,
    col_id    int           NOT NULL       COMMENT '所属院系',
    FOREIGN KEY (col_id) REFERENCES college(col_id) ON UPDATE CASCADE
) COMMENT '教研室';

CREATE TABLE title
(
    tt_id       int            NOT NULL  COMMENT '职称编号' PRIMARY KEY ,
    tt_name     varchar(10)    NOT NULL  COMMENT '职称' UNIQUE
) COMMENT '职称';

DROP TABLE IF EXISTS teacher;
CREATE TABLE teacher
(
    tc_id     int           NOT NULL    COMMENT '教员工号' PRIMARY KEY ,
    tc_name   varchar(10)   NOT NULL    COMMENT '教员姓名' ,
    tc_title  int           NOT NULL    COMMENT '教员职称',
    phone     varchar(11)   NOT NULL    COMMENT '联系方式',
    tr_id     int           NOT NULL    COMMENT '教研室编号',
    FOREIGN KEY (tc_title) REFERENCES title(tt_id) ON UPDATE CASCADE ,
    FOREIGN KEY (tr_id) REFERENCES  teaching_and_research_office(tr_id) ON UPDATE CASCADE
)COMMENT '教员';

CREATE TABLE graduate_student
(
    gs_id    int            NOT NULL    COMMENT '研究生学号' PRIMARY KEY ,
    gs_name  varchar(20)    NOT NULL    COMMENT '研究生姓名',
    gender   int            NOT NULL    COMMENT '性别',
    tc_id    int            NOT NULL    COMMENT '导师',
    col_id   int            NOT NULL    COMMENT '所属院系',
    FOREIGN KEY (tc_id) REFERENCES teacher(tc_id) ON UPDATE CASCADE ,
    FOREIGN KEY (col_id) REFERENCES college(col_id) ON UPDATE CASCADE
) COMMENT '研究生';

CREATE TABLE class
(
    cs_id           int          NOT NULL    COMMENT '班级编号' PRIMARY KEY ,
    cs_number       int          NOT NULL    COMMENT '班级人数',
    cs_instructor   varchar(20)  NOT NULL    COMMENT '辅导员',
    col_id          int          NOT NULL    COMMENT '所属院系',
    FOREIGN KEY (col_id) REFERENCES college(col_id) ON UPDATE CASCADE
)COMMENT '班级';

CREATE TABLE student
(
    stu_id     int          NOT NULL    COMMENT '学号' PRIMARY KEY ,
    stu_name   varchar(20)  NOT NULL    COMMENT '学生姓名',
    gender     int          NOT NULL    COMMENT '性别',
    col_id     int          NOT NULL    COMMENT '所属院系',
    cs_id      int          NOT NULL    COMMENT '班级号',
    FOREIGN KEY (col_id) REFERENCES college(col_id) ON UPDATE CASCADE ,
    FOREIGN KEY (cs_id) REFERENCES class(cs_id) ON UPDATE CASCADE
) COMMENT '学生';

CREATE TABLE course
(
    crs_id         int          NOT NULL    COMMENT '课程编号' PRIMARY KEY ,
    crs_name       varchar(20)  NOT NULL    COMMENT '课程名称',
    crs_point      int          NOT NULL    COMMENT '课程学分',
    crs_date       varchar(30)  NOT NULL    COMMENT '上课时间',
    col_id         int          NOT NULL    COMMENT '所属院系',
    FOREIGN KEY (col_id) REFERENCES college(col_id) ON UPDATE CASCADE
) COMMENT '课程';

DROP TABLE IF EXISTS take_course;
CREATE TABLE take_course
(
    tcrs_id     int             NOT NULL    COMMENT '选课编号' PRIMARY KEY ,
    stu_id      int             NOT NULL    COMMENT '学生学号',
    crs_id      int             NOT NULL    COMMENT '课程编号',
    grade       DECIMAL(3,1)    NOT NULL    COMMENT '成绩',
    FOREIGN KEY (stu_id) REFERENCES student(stu_id) ON UPDATE CASCADE ,
    FOREIGN KEY (crs_id) REFERENCES course(crs_id) ON UPDATE CASCADE
) COMMENT '选修';
